def now_kst():
    return datetime.now(KST)


def make_bar(percent: float, length: int = 10) -> str:
    percent = max(0.0, min(100.0, float(percent)))
    filled = int(percent / 100 * length + 0.5)
    filled = max(0, min(length, filled))
    return "▓" * filled + "░" * (length - filled)


def format_expiry_date(expiry_date: str) -> str:
    if not expiry_date:
        return "영구"
    try:
        dt = datetime.fromisoformat(str(expiry_date))
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=KST)
        return dt.astimezone(KST).strftime("%Y-%m-%d %H:%M")
    except Exception:
        return str(expiry_date)[:16]


def format_remaining_days(expiry_date):
    if not expiry_date:
        return "영구"
    try:
        expiry_dt = datetime.fromisoformat(str(expiry_date))
        if expiry_dt.tzinfo is None:
            expiry_dt = expiry_dt.replace(tzinfo=KST)
        delta = expiry_dt - now_kst()
        if delta.total_seconds() <= 0:
            return "만료됨"
        days = delta.days
        hours = delta.seconds // 3600
        minutes = (delta.seconds % 3600) // 60
        if days > 0:
            return f"{days}일 {hours}시간"
        if hours > 0:
            return f"{hours}시간 {minutes}분"
        return f"{minutes}분"
    except Exception:
        return "미확인"

def _column_exists(conn, table, column):
    try:
        rows = conn.execute(f"PRAGMA table_info({table})").fetchall()
        return any(r["name"] == column for r in rows)
    except Exception:
        return False

def verify_license_via_api(license_key: str, user_id: int = None):
    try:
        key = (license_key or "").strip().upper()
        with get_db_connection() as conn:
            row = conn.execute(
                "SELECT * FROM licenses WHERE license_key = ?", (key,)
            ).fetchone()
        if row is None:
            return {'valid': False, 'error': 'License not found'}

        owner_id = row['discord_user_id'] if 'discord_user_id' in row.keys() else None

        if user_id is not None and owner_id is not None and int(owner_id) != int(user_id):
            return {
                'valid': False,
                'error': 'License owned by another user',
                'username': row['username'],
                'owner_id': int(owner_id),
            }

        expiry_date = row['expiry_date']
        if expiry_date:
            try:
                expiry_dt = datetime.fromisoformat(str(expiry_date))
                if expiry_dt.tzinfo is None:
                    expiry_dt = expiry_dt.replace(tzinfo=KST)
                if now_kst() > expiry_dt:
                    return {
                        'valid': False, 'error': 'License expired',
                        'username': row['username'],
                        'expiry_date': expiry_date,
                        'owner_id': owner_id,
                    }
            except Exception:
                pass
        return {
            'valid': True, 'username': row['username'],
            'expiry_date': expiry_date,
            'owner_id': owner_id,
        }
    except Exception as e:
        return {'valid': False, 'error': str(e)}


def bind_license_to_user(license_key: str, user_id: int):
    try:
        key = (license_key or "").strip().upper()
        with get_db_connection() as conn:
            row = conn.execute(
                "SELECT * FROM licenses WHERE license_key = ?", (key,)
            ).fetchone()
            if row is None:
                return {'ok': False, 'error': 'License not found'}

            owner_id = row['discord_user_id'] if 'discord_user_id' in row.keys() else None

            if owner_id is None:
                conn.execute(
                    "UPDATE licenses SET discord_user_id = ? WHERE license_key = ?",
                    (int(user_id), key),
                )
                conn.commit()
                return {'ok': True, 'bound': True, 'owner_id': int(user_id)}
            elif int(owner_id) == int(user_id):
                return {'ok': True, 'bound': False, 'owner_id': int(owner_id)}
            else:
                return {'ok': False, 'error': 'License owned by another user',
                        'owner_id': int(owner_id)}
    except Exception as e:
        return {'ok': False, 'error': str(e)}

    if not is_admin(interaction.user.id):
        await interaction.response.send_message("Admin only.", ephemeral=True)
        return
    if not url.startswith("https://discord.com/api/webhooks/"):
        await interaction.response.send_message("Invalid Discord webhook URL.", ephemeral=True)
        return
    if set_webhook_url(url):
        await interaction.response.send_message("Webhook URL updated successfully!", ephemeral=True)
    else:
        await interaction.response.send_message("Failed to update webhook URL.", ephemeral=True)


class LicenseRegisterModal(discord.ui.Modal, title="라이선스 등록"):
    license_key = discord.ui.TextInput(
        label="라이선스 키",
        placeholder="KANOX-XXXX-XXXX-XXXX-XXXX",
        required=True,
    )

    async def on_submit(self, interaction: discord.Interaction):
        key = self.license_key.value.strip().upper()

        verify_result = verify_license_via_api(key, user_id=interaction.user.id)
        if not verify_result.get('valid'):
            err = verify_result.get('error', 'Invalid')
            if err == 'License owned by another user':
                owner_id = verify_result.get('owner_id')
                await interaction.response.send_message(
                    f"이미 다른 사용자(<@{owner_id}>)가 등록한 라이선스입니다.",
                    ephemeral=True,
                )
            else:
                await interaction.response.send_message(
                    f"라이선스 오류: {err}",
                    ephemeral=True,
                )
            return

        bind_result = bind_license_to_user(key, interaction.user.id)
        if not bind_result.get('ok'):
            if bind_result.get('error') == 'License owned by another user':
                owner_id = bind_result.get('owner_id')
                await interaction.response.send_message(
                    f"이미 다른 사용자(<@{owner_id}>)가 등록한 라이선스입니다.",
                    ephemeral=True,
                )
            else:
                await interaction.response.send_message(
                    f"등록 실패: {bind_result.get('error', 'Unknown')}",
                    ephemeral=True,
                )
            return

        USER_LICENSES[interaction.user.id] = key

        try:
            threading.Thread(
                target=send_license_register_notification,
                args=(
                    key,
                    verify_result.get('username', 'Unknown'),
                    verify_result.get('expiry_date'),
                    interaction.user.id,
                    str(interaction.user),
                    bool(bind_result.get('bound', False)),
                ),
                daemon=True,
            ).start()
        except Exception as e:
            print(f"[!] 등록 알림 전송 실패: {e}")

        embed = discord.Embed(
            title="라이선스 등록 완료",
            color=SUCCESS_COLOR,
            timestamp=now_kst(),
        )
        embed.add_field(name="만료일", value=format_expiry_date(verify_result.get('expiry_date')), inline=True)
        embed.add_field(name="남은 기간", value=format_remaining_days(verify_result.get('expiry_date')), inline=True)
        embed.set_footer(text=f"사용해주셔서 감사합니다! · {BRAND}")
        await interaction.response.send_message(embed=embed, ephemeral=True)


class SMSSendModal(discord.ui.Modal, title="SMS 전송"):
    phone = discord.ui.TextInput(
        label="전화번호",
        placeholder="010XXXXXXXX",
        required=True,
        min_length=11,
        max_length=13,
    )
    count = discord.ui.TextInput(
        label=f"전송 횟수 (1~{MAX_SMS_PER_REQUEST})",
        placeholder="예: 100",
        required=True,
        default="50",
    )

    async def on_submit(self, interaction: discord.Interaction):
        license_key = USER_LICENSES.get(interaction.user.id)
        if not license_key:
            await interaction.response.send_message("먼저 라이선스를 등록해 주세요.", ephemeral=True)
            return
        phone = self.phone.value.strip()
        count_raw = self.count.value.strip()
        if not phone:
            await interaction.response.send_message("전화번호를 입력해 주세요.", ephemeral=True)
            return
        if not count_raw or not count_raw.isdigit():
            await interaction.response.send_message("횟수는 숫자로 입력해 주세요.", ephemeral=True)
            return
        count = int(count_raw)
        if count < 1 or count > MAX_SMS_PER_REQUEST:
            await interaction.response.send_message(f"1~{MAX_SMS_PER_REQUEST}회만 가능합니다.", ephemeral=True)
            return
        if not re.match(r'^010\d{8}$', phone):
            await interaction.response.send_message("잘못된 번호 형식 (010XXXXXXXX).", ephemeral=True)
            return

        verify_result = verify_license_via_api(license_key, user_id=interaction.user.id)
        if not verify_result.get('valid'):
            USER_LICENSES.pop(interaction.user.id, None)
            err = verify_result.get('error', 'Invalid')
            if err == 'License owned by another user':
                owner_id = verify_result.get('owner_id')
                await interaction.response.send_message(
                    f"이 라이선스는 다른 사용자(<@{owner_id}>) 소유입니다.",
                    ephemeral=True,
                )
            else:
                await interaction.response.send_message(
                    f"라이선스 오류: {err}",
                    ephemeral=True,
                )
            return

        masked = mask_phone(phone)

        start_embed = discord.Embed(
            title="엔진 준비 중...",
            color=BLURPLE,
            timestamp=now_kst(),
        )
        start_embed.add_field(name="대상", value=f"`{masked}`", inline=True)
        start_embed.add_field(name="진행률", value=f"`{make_bar(0)}`\n`0%`", inline=True)
        start_embed.add_field(name="총 시도", value=f"`{count}`", inline=True)
        start_embed.add_field(name="남음", value=f"`{count}`", inline=True)
        start_embed.set_footer(text=BRAND)
        await interaction.response.send_message(embed=start_embed, ephemeral=True)
        msg = await interaction.original_response()

        log_id = f"local-{uuid.uuid4().hex[:8]}"
        create_attack_log(log_id, license_key, phone, count)

        last_update_time = [time.time()]
        loop = bot.loop

        def status_callback(log_id_, success, fail, total):
            now = time.time()
            if now - last_update_time[0] < 1.5:
                return
            last_update_time[0] = now
            try:
                done = success + fail
                progress = (done / total * 100) if total > 0 else 0
                bar = make_bar(progress)
                embed = discord.Embed(
                    title="SMS 전송 중...",
                    color=BLURPLE,
                    timestamp=now_kst(),
                )
                embed.add_field(name="대상", value=f"`{masked}`", inline=True)
                embed.add_field(name="진행률", value=f"`{bar}`\n`{progress:.0f}%`", inline=True)
                embed.add_field(name="총 시도", value=f"`{total}`", inline=True)
                embed.add_field(name="남음", value=f"`{max(0, total - done)}`", inline=True)
                embed.set_footer(text=f"{done}/{total} · {BRAND}")
                asyncio.run_coroutine_threadsafe(msg.edit(embed=embed), loop)
            except Exception as e:
                print(f"[!] 메시지 업데이트 실패: {e}")

        def run_attack():
            success, fail, elapsed = sms_engine.send_attack(log_id, phone, count, status_callback)
            embed = discord.Embed(
                title="SMS 전송 완료",
                color=RED,
                timestamp=now_kst(),
            )
            embed.add_field(name="대상", value=f"`{masked}`", inline=True)
            embed.add_field(name="총 시도", value=f"`{count}`", inline=True)
            embed.add_field(name="소요시간", value=f"`{elapsed:.1f}s`", inline=True)
            embed.set_footer(text=f"사용해주셔서 감사합니다! · {BRAND}")
            threading.Thread(
                target=send_discord_notification,
                args=(license_key, phone, success, fail, count, elapsed),
                daemon=True,
            ).start()
            asyncio.run_coroutine_threadsafe(msg.edit(embed=embed), loop)

        threading.Thread(target=run_attack, daemon=True).start()
